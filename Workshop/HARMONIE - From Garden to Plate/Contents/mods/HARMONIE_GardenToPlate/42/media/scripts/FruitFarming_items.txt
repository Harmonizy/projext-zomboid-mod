    /* Dried pods must be ROASTED before they can be ground. Uses the vanilla
       cooking mechanic (CantEat + IsCookable, as on CannedCarrots): put them in
       an oven, campfire or grill and they become "Cooked" — that is the roast.
       Grinding requires flags[IsCookedFoodItem], so raw pods can never become
       coffee. Roasting is also what detoxifies real Kentucky-coffeetree seeds.
       Window is tighter than vanilla veg (45/70) because coffee scorches. */

    item CoffeePodDried
    {
        Weight = 0.2,

        ItemType = base:food,

        WorldStaticModel = CoffeetreePod_Ground,
        CantEat = true,
        IsCookable = true,
        MinutesToCook = 45,
        MinutesToBurn = 70,
    }
