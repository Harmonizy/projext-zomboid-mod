       Structure mirrors vanilla 42.20.2 CollectSeeds / CollectSunflowerSeeds
       (media/scripts/generated/recipes/recipes_farming.txt): tweezers-or-knife
       tool kept, fruit destroyed, rotten fruit allowed. */

    craftRecipe FF_ExtractAppleSeeds
    {
        timedAction = Making,
        time = 50,
        Tags = InHandCraft;Farming,
        category = Farming,
        inputs
        {
            item 1 tags[base:tweezers;base:dullknife;base:sharpknife] mode:keep flags[MayDegradeLight;Prop1;IsNotDull],
            item 1 [Base.Apple] flags[AllowRottenItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 3 FruitFarming.AppleSeed,
        }
    }

    /* Rice grain chain. Dry/Thresh mirror vanilla DryWheat / ThreshGrain
       (generated/recipes/recipes_farming.txt); drying uses the vanilla drying
       rack via Tags = DryingRackGrain. Seed bootstrap mirrors CollectSeeds.
       FF_PackRice is the one intentional non-vanilla addition: 12 x -5 hunger
       seeds -> one -60 hunger Base.Rice bag, so farmed rice is cookable. */

    craftRecipe FF_CollectRiceSeeds
    {
        timedAction = Making,
        time = 50,
        Tags = InHandCraft;Farming,
        category = Farming,
        inputs
        {
            item 1 tags[base:tweezers;base:dullknife;base:sharpknife] mode:keep flags[MayDegradeLight;Prop1;IsNotDull],
            item 1 [Base.Rice] flags[AllowRottenItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 12 FruitFarming.RiceSeed,
        }
    }

    craftRecipe FF_DryRice
    {
        time = 172800,
        Tags = DryingRackGrain,
        category = Farming,
        overlayStyle = Wheat,
        inputs
        {
            item variable[1:20] [FruitFarming.RiceSheaf] flags[ItemCount] mode:destroy,
        }
        outputs
        {
            item variable[1:20] FruitFarming.RiceSheafDried,
        }
    }

    craftRecipe FF_ThreshRice
